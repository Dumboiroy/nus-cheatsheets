  \section{Asymptomatic Analysis}

  \subsection{Master Theorum}
  \begin{itemize}
    \item $T(n) = aT(\frac{n}{b}) + f(n)$
    \item nLeaves = $n^d$ where $d = \log _b a$ is the critical exponent
    \item Case 1: $f(n) \in O (n^{d-\epsilon}) : \epsilon > 0$ (NR cost << nLeaves)
          \subitem{$\rightarrow T(n) \in \Theta (n^d)$}
          \subitem{NR Cost << nLeaves $\rightarrow$ Most computation: base cases}
    \item Case 2: $f(n) \in \Theta(n^d \log ^k n : k \geq 0)$ (NR cost $\approx$ nLeaves)
          \subitem{$\rightarrow T(n) \in \Theta (n^d \log ^{k+1}n = f(n) \times \log (n))$ = NR Cost x Height}
          \subitem{NR Cost $\approx$ nLeaves $\rightarrow$ Computation shared evenly}
    \item Case 3: $f(n) \in \Omega(n^{d+\epsilon} : \epsilon > 0) \wedge (af(\frac{n}{b}) \leq cf(n)) : c < 1$ (NR cost >> nLeaves)
          \subitem{$\rightarrow T(n) \in \Theta (f(n))$}
          \subitem{NR Cost >> nLeaves $\rightarrow$ Most computation: higher levels}
    \item
          Note that the first condition in Case 3 is redundant. Main(Regularity) condition is that
          the combined subproblem cost $\leq$ current problem non-recurring cost.
  \end{itemize}

  \subsection{Helpers}
  \subsubsection{Stirling's Approx}
  \begin{itemize}
    \item $\log (n!)\approx \Theta (\log n) + \Theta (n \log n) - \Theta (n)$
  \end{itemize}
  \subsubsection{Log rule}
  \begin{itemize}
    \item Log $\log _a b = c \rightarrow b = a^c$
  \end{itemize}
  \subsubsection{AP/GP}
  \begin{itemize}
    \item Where: $a_1$ = first term, $a_n$ = $n$-th term, $d$ = common difference, $r$ = common ratio, $S_n$ = sum of first $n$ terms
  \end{itemize}
  \begin{minipage}{0.5\columnwidth}
    \begin{tabularx}{\linewidth}{X}
      \begin{itemize}
        \item Arithmetic
        \item $a_n = a_1 + (n-1)d$
        \item $S_n = \frac{n}{2}[2a_1+(n-1)d]$
        \item $S_n = \frac{n}{2}(a_1+a_n)$
      \end{itemize}
    \end{tabularx}
  \end{minipage}
  \hfill
  \begin{minipage}{0.5\columnwidth}
    \begin{tabularx}{\linewidth}{X}
      \begin{itemize}
        \item Geometric
        \item $a_n = a_1r^{n-1}$
        \item $S_n = a_1\frac{1-r^n}{1-r}$
        \item $S_\infty = \frac{a_1}{1-r},\quad |r|<1$
      \end{itemize}
    \end{tabularx}
  \end{minipage}


  \section{Dynamic Programming Optimal Substructures}

  \subsection{Fibonacci}
  \begin{itemize}
    \item Input: Integer $n$
    \item Recurrence formula:
    \item $F(n) = \begin{cases} n & \text{if } n \le 1 \\ F(n-1) + F(n-2) & \text{otherwise} \end{cases}$
    \item Recurrence state: $1D$ array or memoization table of size $n+1$
    \item Subproblem dependency: $F(n)$ depends on $F(n-1)$ and $F(n-2)$
    \item{$\rightarrow$ Time Complexity: $\Theta(n)$ with memoization or bottom-up tabulation}
    \item{$\rightarrow$ Space Complexity: $\Theta(1)$ optimized (iterative) or $\Theta(n)$ unoptimized}
  \end{itemize}

  \subsection{Longest Common Subsequence (LCS)}
  \begin{itemize}
    \item Input: Two arrays $A[1..m]$ and $B[1..n]$
    \item Recurrence formula $L(i,j)$:
    \item $L(i, j) = \begin{cases} 0 & \text{if } i = 0 \text{ or } j = 0 \\ L(i-1, j-1) + 1 & \text{if } i, j > 0 \text{ and } a_i = b_j \\ \max(L(i-1, j), L(i, j-1)) & \text{if } i, j > 0 \text{ and } a_i \neq b_j \end{cases}$
    \item Recurrence state: $2D$ table $L[0..m, 0..n]$
    \item Subproblem dependency: $L(i,j)$ depends on diagonal $L(i-1,j-1)$, top $L(i-1,j)$, and left $L(i,j-1)$ cells
    \item{$\rightarrow$ Space Complexity: $\Theta(mn)$ space ($\Theta(\min(m,n))$ with space optimization)}
    \item{$\rightarrow$ Time Complexity: $\Theta(mn)$ time}
  \end{itemize}

  \subsection{0/1-Knapsack}
  \begin{itemize}
    \item Input: $n$ items described as $(weight, value)$ pairs: $\{(w_1, v_1), (w_2, v_2), \dots, (w_n, v_n)\}$ and capacity $W$
    \item Recurrence formula $K(i, j)$ (the value of an optimal solution for the instance $\{(w_1, v_1), \dots, (w_i, v_i)\}$ and capacity $j$):
    \item $K(i, j) = \begin{cases} -\infty & \text{if } j < 0 \\ 0 & \text{if } i = 0 \text{ and } j \ge 0 \\ \max(v_i + K(i-1, j-w_i), K(i-1, j)) & \text{otherwise} \end{cases}$
    \item Recurrence state: $2D$ table $K[0..n, 0..W]$
    \item Subproblem dependency: $K(i,j)$ depends on taking item $i$ (if $j \ge w_i$) or skipping item $i$
    \item{$\rightarrow$ Time Complexity: $\Theta(nW)$ pseudopolynomial time}
    \item{$\rightarrow$ Space Complexity: $\Theta(nW)$ space ($\Theta(W)$ with rolling array)}
  \end{itemize}

  \subsection{Coin-Change Problem}
  \begin{itemize}
    \item Goal: Find the minimum number of coins of denominations $\{d_1, d_2, \dots, d_k\}$ that add up to $n$ cents
    \item Recurrence formula $M(n)$ (fewest number of coins needed to change $n$ cents):
    \item $M(n) = \begin{cases} \infty & \text{if } n < 0 \\ 0 & \text{if } n = 0 \\ 1 + \min_{i \in [1..k]} M(n - d_i) & \text{otherwise} \end{cases}$
    \item Recurrence state: $1D$ array $M[0..n]$
    \item Subproblem dependency: $M(n)$ depends on subproblems of smaller remaining amounts $n - d_i$
    \item{$\rightarrow$ Time Complexity: $\Theta(n \cdot k)$ time ($\Theta(n)$ if $k$ is treated as constant)}
    \item{$\rightarrow$ Space Complexity: $\Theta(n)$ space ($\Theta(k)$ per state or $\Theta(n)$ aux space)}
  \end{itemize}

  \subsection{Longest Increasing Subsequence (LIS)}
  \begin{itemize}
    \item Input: Given an array $A[1..n]$ with $n$ integers, plus two sentinels $A[0] = -\infty$ and $A[n+1] = \infty$
    \item Recurrence formula $L(j)$ (length of the longest strictly increasing subsequence of $A[1..j]$ that ends at index $j$, for $j \in [0..n+1]$, computing $L(n+1) - 1$):
    \item $L(j) = \begin{cases} 0 & \text{if } j = 0 \\ 1 + \max_{i \in [0..j-1] \text{ and } A[i] < A[j]} L(i) & \text{if } j > 0 \end{cases}$
    \item Recurrence state: $1D$ array $L[0..n+1]$ where $L[j]$ is length of LIS ending at index $j$
    \item Subproblem dependency: $L(j)$ depends on all valid preceding smaller elements where $A[i] < A[j]$
    \item{$\rightarrow$ Time Complexity: $\Theta(n^2)$ time ($\Theta(n \log n)$ with binary search / patience sorting)}
    \item{$\rightarrow$ Space Complexity: $\Theta(n)$ space}
  \end{itemize}
