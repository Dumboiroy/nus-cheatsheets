    \subsection{Characterizing the Environment}

    \begin{enumerate}
        \item \keyword{Fully observable}{(vs. Partially) Agent's sensors can access complete state of env. all the time}
        \item \keyword{Deterministic}{(vs. Stochastic) Next state of env. is determined by \textbf{current state} and \textbf{action executed by agent}}
              \begin{itemize}
                  \item \keyword{Strategic}{If env. is deterministic except for actions of other agents}
              \end{itemize}
        \item \keyword{Episodic}{(vs. Sequential) Agent's experience is divided into atomic \textbf{episodes}, where each episode includes perceiving and an action, and \textbf{action depends on episode} itself}
        \item \keyword{Static}{(vs. Dynamic) Env. is unchanged while agent is deciding}
              \begin{itemize}
                  \item \keyword{Semi}{Time does not affect env., but affects performance score}
              \end{itemize}
        \item \keyword{Discrete}{(vs. Continuous) Discrete num. of percepts and actions}
        \item \keyword{Single Agent}{(vs. Multi-agent) Agent operating by itself in an env.}
    \end{enumerate}

    \subsection{Performance Measurement}

    \begin{itemize}
        \item \keyword{Correctness}{Correct if $\hat{y} = y$}
        \item \keyword{Accuracy}{$\frac{1}{m} \sum_{j = 1}^{m} (\hat{y_{j}} = y_{j})$}
    \end{itemize}

    \begin{multicols*}{2}

        \begin{itemize}
            \item Confusion Matrix:
        \end{itemize}

        \includegraphics[scale=0.2]{confusion-matrix}

        \begin{itemize}
            \item Accuracy = $\frac{TP + TN}{TP + FN + FP + TN}$
            \item \keyword{Precision}{$\frac{TP}{TP + FP}$} How precise are positive predictions?
            \item \keyword{Recall}{$\frac{TP}{TP + FN}$} How many actual positives are predicted?
            \item \keyword{F1 Score}{$\frac{2}{1 / P + 1 / R}$} Harmonic mean of precision and recall
        \end{itemize}

    \end{multicols*}

    \begin{itemize}
        \item Type I Error: FP, Type II Error: FN
        \item FP Rate = $\frac{FP}{FP + TN}$ TP Rate = $\frac{TP}{TP + FN}$
    \end{itemize}

    \subsection{Pruning}

    \begin{itemize}
        \item Motivation: DT overfits to training set, but performs poorly on test set
        \item Occam's Razor: Simple hypothesis preferred
        \item \keyword{Pruning}{Ignores outliers, which reduces overfitting}
              \begin{itemize}
                  \item Idea: Go with the majority of T/Fs
                  \item E.g. Min-sample, Max-depth
              \end{itemize}
    \end{itemize}


    \section{Regularization}
    \subsection{Overfitting}
    \begin{itemize}
        \item Model learns training data too well $\Rightarrow$ high training accuracy, poor generalization
        \item Caused by: model complexity $>$ ground truth complexity
    \end{itemize}
    \subsection{Bias-variance Tradeoff}
    \begin{itemize}
        \item High bias models are too simple and cannot capture patterns well
        \item High variance models are too complex and are overfitted to training data
        \item We use regularization on high variance models to get the model to learn ground truth patterns rather than memorize the data
        \item a good model has balanced bias and variance in a way that fits the task
    \end{itemize}

    \subsection{$L_p$-norm}
    Vector size measure ($p \geq 1$):
    \[\|w\|_p = \left(\sum_{j=0}^{d} |w_j|^p\right)^{1/p}\]
    \begin{itemize}
        \item $L_1$: $\|w\|_1 = \sum_j |w_j|$ (Manhattan)
        \item $L_2$: $\|w\|_2 = \sqrt{\sum_j w_j^2}$ (Euclidean)
        \item $L_\infty$: $\|w\|_\infty = \max_j |w_j|$ (Max)
    \end{itemize}

    \subsection{Regularization}
    Adds penalty to prevent model complexity: $J_{\text{reg}}(w) = J(w) + \lambda R(w)$
    \begin{itemize}
        \item $\lambda \geq 0$: regularization strength
        \item $R(w)$: penalty function (norm-based)
        \item Observation: overfitting $\Leftrightarrow$ large weight magnitudes
    \end{itemize}

    \subsubsection{Common Regularizers}
    \begin{itemize}
        \item Lasso: $J(w) + \lambda \|w\|_1$ (sparse solutions)
        \item Ridge: $J(w) + \lambda \|w\|_2$ (weight shrinkage)
        \item Elastic Net: $J(w) + \lambda_1 \|w\|_1 + \lambda_2 \|w\|_2$
    \end{itemize}

    \subsection{Ridge Regression ($L_2$ Regularization)}
    Closed-form: $w = (X^TX + \lambda I)^{-1}X^Ty$

    \begin{itemize}
        \item Invertibility: $X^TX + \lambda I$ invertible $\forall \lambda > 0$
        \item Unique solution exists
        \item Shrinks weights uniformly, none become zero
    \end{itemize}

    \subsection{Lasso Regression ($L_1$ Regularization)}
    \begin{itemize}
        \item Induces sparsity: some weights exactly zero
        \item Threshold rule: contribution to MSE reduction $<$ penalty $\lambda$ $\Rightarrow$ weight set to 0
        \item Feature selection automatic
    \end{itemize}

    \section{Kernel \& Feature Transformation}
    %new info start%
    \subsection{The Kernel Function:}
    \[K(u,v) = u \cdot v \quad \text{(dot product of two input vectors)}\]
    where $u, v \in \mathbb{R}^d$ are two input points, returns a scalar.
    Kernels = similarity function in feature space, not distance.

    \subsection{The Feature Map:}
    \[\phi(x): \mathbb{R}^d \to \mathbb{R}^M \quad \text{(transforms input into feature space)}\]
    For the linear kernel: $\phi(x) = x$ (identity transformation)

    \subsection{Fundamental Kernel-Feature Relationship:}
    \[K(u,v) = \phi(u) \cdot \phi(v)\]
    The kernel equals the dot product of transformed feature vectors.

    \subsubsection{examples}
    \text{Kernel Formulas:}
    $K_{new}(u,v) = 4 \times K(u,v) = 4(u \cdot v) = (2u \cdot 2v)$
    Scaling factor: 4, OR \newline
    $K(u,v) = K(u,v) = \log(u) \cdot \log(v) = \log(u)^T \cdot \log(v) = \log(u_1)^T\log(v_1) \cdot \log(u_2)^T\log(v_2)$

    \textbf{The New Feature Map (Unknown):}
    \[\phi_{new}(x) = 2x \]
    We need to find the feature map corresponding to $K_{new}$.

    \subsection{The Prediction Formula (Kernel Methods)}
    \[h_a(x) = \sum_{i=1}^{N} a_i K(x^{(i)}, x)\]
    where:
    \begin{itemize}
        \item $x^{(i)}$ = $i$-th training point
        \item $a_i$ = learned weight for training point $i$
        \item $K(x^{(i)}, x)$ = kernel between training point $i$ and new point $x$
        \item $N$ = total number of training points
    \end{itemize}

    \subsection{Feature Transformation}
    \[x \in \mathbb{R}^d \xrightarrow{\phi} \phi(x) \in \mathbb{R}^M \quad (M \geq d)\]

    Polynomial degree $k$: $\binom{d+k}{k}$ features
    \begin{itemize}
        \item 2 features, degree 2: $\binom{2+2}{2} = 6$ features
              % \item 3 features, degree 2: 10 features
              % \item 100 features, degree 6: $\sim 1.6B$ features
              % \item Problem: computationally intractable for large $d, k$
    \end{itemize}

    \subsection{Linear Combination of Vectors}
    $v = \sum_{i=1}^{n} a_i x^{(i)}, \quad a_i \in \mathbb{R}$

    Geometric interpretation: $v$ lies in span of $\{x^{(1)}, \ldots, x^{(n)}\}$

    \subsection{Dual Formulation: Key Theorem}
    \textbf{Theorem:} Optimal weights for MSE + $L_2$ regularization:
    $w = \sum_{j=1}^{n} \alpha_j x^{(j)} = X^T\alpha$

    where $\alpha \in \mathbb{R}^n$ (coefficient vector)

    \begin{itemize}
        \item Weights = linear combination of training vectors
        \item Number of parameters: $n$ (training samples), not $d$ (features)
        \item $\Rightarrow$ works for high-dimensional/infinite feature spaces
        \item Foundation for kernel methods, SVM
    \end{itemize}

    \subsection{Normal Equation (Ridge)}
    $X^T(Xw - y) + n\lambda w = 0 \Rightarrow X^T(Xw-y) + n\lambda Iw = 0$

    Closed-form: $w = (X^TX + n\lambda I)^{-1}X^Ty$

    Using Woodbury matrix identity:
    \[w = X^T(XX^T + n\lambda I)^{-1}y = X^T\alpha\]

    where $\alpha = (XX^T + n\lambda I)^{-1}y$




    \section{Neural Networks}

    \subsection{Neuron Model}
    \textbf{Basic neuron:} input features $x_0, \ldots, x_d$ with weights $w_0, \ldots, w_d$
    $z = \sum_{j=0}^{d} w_j x_j \quad \text{(aggregate)}$ \newline
    $a = g(z) \quad \text{(activate)} \quad \text{output: } \hat{y} = a$ \newline
    $g$: activation function (non-linear). $w_0$ (bias term, $x_0=1$).

    \subsection{Common Activation Functions}
    \begin{center}
        \begin{tabular}{lll}
            \textbf{Sigmoid}    & $g(z)=\sigma(z)=\dfrac{1}{1+e^{-z}}$           & $g'(z)=\sigma(z)\,(1-\sigma(z))$                 \\
            \textbf{Identity}   & $g(z)=z$                                       & $g'(z)=1$                                        \\
            \textbf{Tanh}       & $g(z)=\tanh(z)=\dfrac{e^z-e^{-z}}{e^z+e^{-z}}$ & $g'(z)=1-\tanh^2(z)$                             \\
            \textbf{ReLU}       & $g(z)=\max(0,z)$                               & $g'(z)=\begin{cases}0,& z<0\\1,& z>0\end{cases}$ \\
            \textbf{Leaky ReLU} & $g(z)=\max(az,z)\ (a>0)$                       & $g'(z)=\begin{cases}a,& z<0\\1,& z>0\end{cases}$ \\
        \end{tabular}
    \end{center}

    \subsection{Neural Networks}
    \includegraphics[width=0.6\linewidth]{nn}

    %     \item Node $v \in V$: contains function $h_v$ parameterized by $w^{[v]}$
    %     \item Edge $e \in E$: data flow

    \subsection{Feed-Forward Neural Networks (FFNNs)}
    Special case: computation graph is DAG (input $\to$ output, no cycles).
    \begin{itemize}
        \item \textbf{Layer:} group of parallel neuron models $l_w: X_{in} \to X_{out}$
        \item \textbf{Fully-connected (FC) layer:} all inputs connected to all neurons
        \item \textbf{Depth:} number of sequential layers (excl. input)
        \item \textbf{Width:} number of neurons per layer
    \end{itemize}

    \subsection{FCNN: Matrix Representation}
    \includegraphics[width=\linewidth]{nn-calculations}
    \textbf{Single layer:} $x \in \mathbb{R}^d \to \hat{y} = g(W^T x)$ where $W \in \mathbb{R}^{d \times m}$ ($m$ outputs)
    $\hat{y} = g\left(W^T x\right)$
    \textbf{Multiple layers:} composition of matrix operations
    $\hat{y} = g^{[L]}\left(W^{[L]^T} g^{[L-1]}\left(\cdots g^{[1]}\left(W^{[1]^T} x\right) \cdots\right)\right)$

    \textbf{Or Simply:}\newline
    $z^{[l]} = W^{[l]} \cdot a^{[l-1]} + b^{[l]}$ and $a^{[l]} = g^{[l]} \cdot z^{[l]}$\newline
    $b^{[l]}$: bias at layer $l$ |
    $W^{[l]}$: Weights at layer $l$ |
    $z^{[l]}$: pre-activation value at layer $l$ |
    $a^{[l]}$: post-activation input at layer $l$, \textbf{input of layer $l+1$}
    \subsection{Gradient Formula}
    $\frac{\partial \mathcal{L}}{\partial W^{[l]}} = \delta^{[l]} (a^{[l-1]})^T$\newline

    \subsection{Derivatives of Common Loss Functions}
    Mean Squared Error (MSE):
    $L = \frac{1}{2n}(y - \hat{y})^2$\newline
    $\frac{\partial L}{\partial \hat{y}} = \frac{1}{n} \cdot (y - \hat{y}) \cdot (-1) = -\frac{1}{n}(y - \hat{y}) \text{ or } \frac{1}{n}(\hat{y} - y)$\newline
    Loss range: $L \in [0, \infty)$ (always non-negative), Derivative range: $\frac{\partial L}{\partial \hat{y}} \in (-\infty, \infty)$ (any real
    number)\newline

    Binary Cross-Entropy (BCE):
    $L = -y \log(\hat{y}) - (1-y)\log(1-\hat{y})$\newline
    $\frac{\partial L}{\partial \hat{y}} = -\frac{y}{\hat{y}} + \frac{1-y}{1-\hat{y}}$\newline
    Loss range: $L \in (0, \infty)$ (approaches 0 as $\hat{y} \to y$, $\infty$ as $\hat{y} \to
        1-y$),
    Derivative range: $\frac{\partial L}{\partial \hat{y}} \in (-\infty, -1) \cup (1, \infty)$
    \quad (negative when $y=1$, positive when $y=0$)\newline


    Absolute Error:
    $L = |y - \hat{y}|$\newline
    $\frac{\partial L}{\partial \hat{y}} = -\text{sign}(y - \hat{y})$\newline
    Loss range: $L \in [0, \infty)$ (non-negative),
    Derivative range: $\frac{\partial L}{\partial \hat{y}} \in \{-1, 0, 1\}$ (sign function; 0 at
    $y = \hat{y}$)\newline

    \subsection{The Error Signal: $\delta^{[l]}$}
    $\delta^{[l]} = \frac{\partial \mathcal{L}}{\partial z^{[l]}}$
    \begin{itemize}
        \item Gradient of loss w.r.t pre-activation at layer $l$.
        \item “how much each neuron at layer $l$ contributed to the final error.”
        \item Controls which direction and magnitude of change of weights \& biases
        \item \textbf{Loss function w.r.t update in weights at layer $l$}
        \item For each layer $l$, we look at:
              $$
                  \frac{\partial \mathcal{L}}{\partial W^{[l]}}
                  = \delta^{[l]} (a^{[l-1]})^T
              $$
        \item Gradient magnitude depends on $\delta^{[l]}$.
        \item \textbf{Backprop structure (key mechanism)}
        \item
              $$
                  \delta^{[l]} = (W^{[l+1]})^T \delta^{[l+1]} \odot \sigma'(z^{[l]})
              $$
        \item Unrolling this through N layers:
              $$
                  \delta^{[1]} =
                  \left(\prod_{k=1}^{N-1} (W^{[k+1]})^T \right)\delta^{[N]}
                  \odot
                  \left(\prod_{k=1}^{N-1} \sigma'(z^{[k]}) \right)
              $$
        \item if $N$ is large, gradient vanishes at early layers
    \end{itemize}
    \subsection{Dying ReLU}
    \begin{enumerate}
        \item Error signal formula:
              $
                  \delta^{[l]} = \left(W^{[l+1]}\right)^T \delta^{[l+1]} \odot g'\!\left(z^{[l]}\right)
              $

        \item For ReLU:
              $
                  g'\!\left(z^{[l]}\right) =
                  \begin{cases}
                      1 & z^{[l]} > 0   \\
                      0 & z^{[l]} \le 0
                  \end{cases}
              $

        \item If $z^{[l]} \le 0$, then the error signal at that neuron becomes:
              $
                  \delta^{[l]} = 0
              $

        \item That neuron does not contribute to backpropagation (no gradient flows through it).
    \end{enumerate}
    \begin{itemize}
        \item Leaky ReLU fixes dying ReLU by ensuring $g'(z) = \alpha > 0$ even when $z \le 0$, so gradients do not become zero.
        \item This keeps $\delta^{[l]} \neq 0$, allowing weights to keep updating and preventing neurons from becoming permanently inactive.
    \end{itemize}


    \subsection{Universal Approximation (Informal)}
    Single \textbf{hidden} layer FCNN with sufficient width + non-linear activation $\implies$ can approximate any continuous function on bounded domain.
    \begin{itemize}
        \item Sigmoid, Tanh, ReLU: all valid activations
        \item Why multiple layers then? Computational efficiency, expressiveness
    \end{itemize}

    \subsection{Regression Tasks}
    \textbf{Single output:} $y \in \mathbb{R}$ (temperature prediction)
    \begin{itemize}
        \item Predictor: single neuron with \textbf{identity activation}
        \item Loss: $J_{\text{MSE}}(w) = \frac{1}{2n}\sum_{i=1}^{n}(\hat{y}^{(i)} - y^{(i)})^2$
    \end{itemize}
    \textbf{Multi-output:} $y \in \mathbb{R}^m$ (temp + humidity)
    \begin{itemize}
        \item Predictor: $m$ neurons with identity activation
        \item Loss: $J_{\text{MSE}}(w) = \frac{1}{2n}\sum_{i=1}^{n}\sum_{j=1}^{m}(\hat{y}_j^{(i)} - y_j^{(i)})^2$
    \end{itemize}

    \subsection{Binary Classification}
    $y \in \{0,1\}$ (rain or not)
    \begin{itemize}
        \item Predictor: single neuron with \textbf{sigmoid activation}
        \item $\hat{y} = \sigma(z) \in [0,1]$ (prob. of class 1)
        \item Loss: $J_{\text{BCE}}(w) = \frac{1}{n}\sum_{i=1}^{n}\text{BCE}(y^{(i)}, \hat{y}^{(i)})$
    \end{itemize}

    \subsection{Multi-class Classification}
    $y \in \{1, \ldots, K\}$ (rain intensity: drizzle, light, heavy)
    \begin{itemize}
        \item Predictor: $K$ neurons with \textbf{softmax activation}
        \item Softmax: $\text{softmax}(z)_i = \frac{e^{z_i}}{\sum_{j=1}^{K}e^{z_j}} \in [0,1]$ (prob. class $i$)
        \item Loss: $J_{\text{CCE}}(w) = -\frac{1}{n}\sum_{i=1}^{n}\sum_{j=1}^{K}y_j^{(i)}\log(\hat{y}_j^{(i)})$ (categorical cross-entropy)
    \end{itemize}

    \subsection{One-Hot Encoding}
    Represent categorical labels as vectors: each class $\to$ vector with 1 in position of class, 0 elsewhere. Problem is high dimensionality and not useful representation of data.
    \begin{center}
        \begin{tabular}{c|c}
            Class   & Vector    \\
            \hline
            Drizzle & $[1,0,0]$ \\
            Light   & $[0,1,0]$ \\
            Heavy   & $[0,0,1]$
        \end{tabular}
    \end{center}

    \subsection{Multi-label Classification}
    $y \in \{0,1\}^m$ (rain intensity AND cyclone risk independently)
    \begin{itemize}
        \item Predictor: $m$ neurons with \textbf{sigmoid activation} (independent outputs)
        \item Loss: sum of binary cross-entropy for each label
    \end{itemize}

    \subsection{Learning via Gradient Descent}
    Minimize loss $J(w)$ by iterative updates:
    $$w \leftarrow w - \alpha \nabla_w J(w)$$
    $\alpha$: learning rate. $\nabla_w J(w)$: gradient computed via \textbf{backpropagation}.

    \subsection{Backpropagation}
    Algorithm: efficiently compute gradients $\frac{\partial J}{\partial w^{[l]}}$ via chain rule.
    \begin{itemize}
        \item Forward pass: compute outputs layer-by-layer
        \item Backward pass: compute gradients from output to input
        \item Complexity: $O(n_{\text{params}})$ (same as forward pass)
    \end{itemize}

    \subsection{Data Format}
    Dataset $D = \{(x^{(1)}, y^{(1)}), \ldots, (x^{(n)}, y^{(n)})\}$:
    \begin{itemize}
        \item Features: $x^{(i)} \in \mathbb{R}^d$ (real-valued)
        \item Targets: $y^{(i)}$ (task-dependent: scalar, vector, categorical, multi-label)
    \end{itemize}

    \section{Convolutional Neural Networks (CNNs)}
    \textbf{FCNN limitations with images:}
    \begin{itemize}
        \item Parameter explosion: 600×600 RGB image $\to$ 1.08M parameters for single hidden layer
        \item Loss of spatial structure: treats pixels as independent features
        \item Solution: exploit local structure via convolution
    \end{itemize}

    \subsection{Convolution Operation}
    \textbf{Kernel/Filter:} small matrix (e.g., 3×3) slides over input, computing element-wise products + sum.
    \begin{itemize}
        \item Input: $H \times W \times C$ (height, width, channels)
        \item Filter: $K \times K \times C$ (shared weights)
        \item Output: $(H-K+1) \times (W-K+1)$ per filter
        \item \textbf{Stride:} step size of kernel movement (stride 2 $\implies$ half spatial dimension)
        \item \textbf{Padding:} add zeros around input to preserve spatial dimension
        \item \textbf{Formula for Output of CNN Layer (input=$H \times W \times C$, Kernel=$h \times w \times c$, stride = s, padding = p)}
        \item $$H_{out} = \left\lfloor \frac{H - h + 2p}{s} \right\rfloor + 1, \quad W_{out} = \left\lfloor \frac{W - w + 2p}{s} \right\rfloor + 1, \quad C_{out} = n_{\text{filters}}$$
              Output shape: $H_{out} \times W_{out} \times C_{out}$
    \end{itemize}

    \subsection{Convolutional Layer}
    $m$ filters + bias $\to$ $m$ output feature maps. Learnable parameters: $m \cdot (K \cdot K \cdot C + 1)$ (independent of input size).
    \begin{itemize}
        \item \textbf{Activation:} ReLU applied per-element
        \item \textbf{Receptive field:} region of input affecting output neuron (grows with network depth)
        \item \textbf{Receptive field formula: }
              $r_i = r_{i-1} + (k_i - 1)\, j_{i-1}, \quad j_i = j_{i-1} \cdot s_i$
              $
                  k_i=\text{kernel size at layer } i\\
                  s_i=\text{stride at layer } i,\\
                  j_i=\text{jump at layer } i,\\
                  r_i=\text{dim at layer } i,\\
                  r_0, j_0=1
              $
        \item if kernel height $\neq$ width, can use the same formula, calculate separately.

    \end{itemize}

    \subsection{Pooling Layers}
    Reduce spatial dimensions, aggregate information:
    \begin{itemize}
        \item \textbf{Max pooling:} $\max(2 \times 2)$ window $\to$ single value (translation invariance)
        \item \textbf{Average pooling:} mean of window (smooth aggregation)
        \item No learnable parameters
        \item Stride typically matches pool size
    \end{itemize}

    \subsection{CNN Architecture}
    Typical pattern: conv $\to$ ReLU $\to$ pool $\to$ repeat. Final layers: flatten + fully-connected for classification.
    \begin{itemize}
        \item Early layers: detect low-level features (edges, colors)
        \item Deep layers: detect high-level features (shapes, objects)
        \item Hierarchical feature extraction via receptive field growth
    \end{itemize}

    \subsection{Sequential Data Representation}
    \textbf{Types:}
    \begin{itemize}
        \item Video: 4D tensor $(T, H, W, C)$ (time, height, width, channels)
        \item Audio: 1D sequence (waveform samples)
        \item Text: sequence of words (variable length)
    \end{itemize}

    \subsection{Recurrent Neural Networks (RNNs)}
    Process sequential data by maintaining hidden state:
    $$h^{[t]} = \tanh(W_h h^{[t-1]} + W_x x^{[t]} + b)$$
    $$\hat{y}^{[t]} = W_y h^{[t]} + b_y$$
    \begin{itemize}
        \item $h^{[t]}$: hidden state at time $t$
        \item $x^{[t]}$: input at time $t$
        \item Shared parameters $(W_h, W_x, W_y)$ across timesteps
    \end{itemize}

    \subsection{RNN Variants}
    \begin{itemize}
        \item \textbf{Bidirectional RNN:} process sequence forward + backward, concatenate $h^{[t]}_{\text{fwd}}$ and $h^{[t]}_{\text{bwd}}$ for output
        \item \textbf{Stacked RNN:} multiple RNN layers (output $h^{[t]}$ of layer $l$ $\to$ input to layer $l+1$)
        \item \textbf{Cardinality:} one-to-many, many-to-one, many-to-many: $s$-to-$o$  where $s$ is number of steps and $o$ is number of output. note that output can also be a sequence
    \end{itemize}

    \subsection{RNN Bottlenecks}
    \begin{itemize}
        \item \textbf{Sequential computation:} cannot parallelize over timesteps $\implies$ slow training/inference
        \item \textbf{Memory decay:} hidden state weakly depends on distant past (vanishing/exploding gradients during backprop)
        \item Long-term dependencies: hard to capture with tanh activation
    \end{itemize}

    \subsection{Text Representation}
    \textbf{One-hot encoding:} vocabulary size $V$ $\to$ each word = vector with 1 in position, 0 elsewhere ($V$-dimensional).
    \begin{itemize}
        \item Issue: sparse, no semantic similarity captured
        \item Solution: \textbf{embedding layer} learns dense representation
    \end{itemize}

    \subsection{Embedding Layer}
    Learnable lookup table $E \in \mathbb{R}^{V \times d}$ ($V$ words, $d$ dimensions):
    \begin{itemize}
        \item Input: word index $i$
        \item Output: $E[i, :] \in \mathbb{R}^d$ (dense vector)
        \item Learns semantic similarity: similar words $\to$ nearby embeddings
        \item Parameters: $V \cdot d$ (learned via gradient descent)
    \end{itemize}

    \subsection{Attention Mechanism}
    Problem: RNN hidden state becomes bottleneck. Solution: allow model to look at \textbf{all} previous timesteps, not just last hidden state.
    \begin{itemize}
        \item Query vector: "what am I looking for?"
        \item Key vector: "what information do I have?"
        \item Value vector: "what should I extract?"
        \item Relevance: $\text{score}(q, k) = q^T k$ (dot product)
    \end{itemize}

    \subsection{Self-Attention: Query, Key, Value}
    \includegraphics[width=0.6\linewidth]{self-attention-qkv}
    For timestep $t$, transform hidden state $h^{[t]}$ via learnable matrices:
    $$q^{[t]} = W_q h^{[t]}, \quad k^{[t]} = W_k h^{[t]}, \quad v^{[t]} = W_v h^{[t]}$$
    Compute relevance of all timesteps to current query:
    $$\text{scores} = Q K^T \in \mathbb{R}^{T \times T} \quad \text{(where } Q = [q^{[1]}, \ldots, q^{[T]}], K = [k^{[1]}, \ldots, k^{[T]}])$$
    Apply softmax + weighted sum of values:
    $$\text{attention}(Q, K, V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right) V$$
    \begin{itemize}
        \item $\frac{1}{\sqrt{d_k}}$ scaling: prevent dot products from exploding
        \item Output: context vector for each timestep (attended information from all timesteps)
    \end{itemize}

    \subsection{Self-Attention Advantages}
    \begin{itemize}
        \item Parallel computation: all timesteps processed simultaneously (no sequential loop)
        \item Long-term dependencies: direct connections between all pairs of timesteps (no memory decay)
        \item Interpretability: attention weights show which words attended to
    \end{itemize}

    \section{Unsupervised Learning with NN}
    \subsection{Supervised Learning Bottleneck}
    \begin{itemize}[leftmargin=*]
        \item \textbf{Supervision Signal:} Labels $y^{(i)}$ explicitly compared $\rightarrow$ precise error correction
        \item \textbf{Annotation Cost:} Time + Money + Domain expertise (e.g., doctors labeling medical scans)
        \item \textbf{Reality:} Vast majority of data (text, images, video) $\rightarrow$ no human annotations
        \item \textbf{Challenge:} How to leverage unlabeled data?
    \end{itemize}

    \subsection{Unsupervised Learning Fundamentals}
    \textbf{Definition:} Train on only inputs $\{x^{(i)}\}$, no human labels. Generate supervision from data itself.

    \textbf{Three-Stage Framework:}
    \begin{enumerate}
        \item \textbf{Pretext Task:} Auxiliary task with auto-generated pseudo-labels
        \item \textbf{Self-Supervised Pre-Training:} Model solves pretext task on unlabeled data
        \item \textbf{Supervised Fine-Tuning:} Transfer learned features to downstream task
    \end{enumerate}

    \subsection{Self-Supervised Learning Framework}
    \textbf{Design New Task (Pretext Task):}
    \begin{itemize}[leftmargin=*]
        \item Formulate task: targets (pseudo-labels) auto-generated from unlabeled input
        \item Generate supervision directly from data itself, without human annotation
        \item Task Requirement: must force model to capture useful features \& underlying structures
        \item Train model to solve new task $\rightarrow$ model supervises itself
    \end{itemize}

    \subsection{Rotation Prediction Pretext Task}
    \textbf{Setup:} Given unlabeled images $\{x^{(i)}\}$

    \textbf{Transformation:} For each image $x^{(i)}$, generate 4 rotated versions:
    \begin{itemize}[leftmargin=*]
        \item $j=0$: rotate $0°$ $\rightarrow$ $x^{(i)[0]}$ ... Total rotated images: $4n_1$
              % \item $j=1$: rotate $90°$ $\rightarrow$ $x^{(i)[1]}$
              % \item $j=2$: rotate $180°$ $\rightarrow$ $x^{(i)[2]}$
              % \item $j=3$: rotate $270°$ $\rightarrow$ $x^{(i)[3]}$
              % \item 
    \end{itemize}

    \textbf{Pseudo-label Generation:} One-hot encode rotation angle $y^{(i)[j]} \in \mathbb{R}^4$
    \begin{itemize}[leftmargin=*]
        \item $j=0$: $y^{(i)[0]} = [1,0,0,0]^T$ ... 4 classes defined by rotation angle
              % \item $j=1$: $y^{(i)[1]} = [0,1,0,0]^T$
              % \item $j=2$: $y^{(i)[2]} = [0,0,1,0]^T$
              % \item $j=3$: $y^{(i)[3]} = [0,0,0,1]^T$
              % \item 
    \end{itemize}

    \textbf{Architecture:} Feature Extractor + Predictor
    \begin{itemize}[leftmargin=*]
        \item \textbf{Feature Extractor:} Conv layer $\rightarrow$ Pooling layer $\rightarrow$ Conv layer $\rightarrow$ Pooling layer $\rightarrow$ Flatten
        \item \textbf{Predictor:} FC layers + 4 outputs $\hat{y}_1, \hat{y}_2, \hat{y}_3, \hat{y}_4$
        \item \textbf{Loss:} Cross-Entropy between $\hat{y}^{(i)[j]}$ and $y^{(i)[j]}$
    \end{itemize}

    \textbf{Why Rotation?} Understanding image content requires:
    \begin{itemize}[leftmargin=*]
        \item Object orientation: recognizing what is ``upright''
        \item Shape \& structure: edges, contours, pose
        \item $\Rightarrow$ Learned features useful for downstream tasks like image classification
    \end{itemize}

    \subsection{Transfer Learning \& Fine-Tuning}
    \textbf{Downstream Task:} Binary image classification (e.g., bird vs. non-bird)

    \textbf{Architecture:}
    \begin{itemize}[leftmargin=*]
        \item \textbf{Feature Extractor:} Reuse from rotation model, initialize with \textbf{pretrained weights}
        \item \textbf{New Predictor:} Task-specific FC layer (1 output for binary), initialize with \textbf{random weights}
    \end{itemize}

    \textbf{Fine-Tuning Strategies:}
    \begin{enumerate}
        \item Update every single weight in entire model
        \item Update subset of model weights (e.g., only predictor, or only later layers)
    \end{enumerate}

    \textbf{Loss:} Binary Cross-Entropy $\text{BCE}(\hat{y}^{(i)}, y^{(i)})$ where $y^{(i)} \in \{0,1\}$

    \subsection{Linear \& Logistic Regression Cheatsheet}

    \subsubsection{Linear Regression Basics}
    Model: $h_w(x) = w_0 + w_1 x_1$ (bias $w_0$, or use dummy $x_0=1$).\\
    MSE Loss: $L_{MSE}(w) = \frac{1}{2n}\sum_{i=1}^n (h_w(x^{(i)}) - y^{(i)})^2$

    \textbf{Example (6A):} $w=[2,1]^\top$, data: $(0,0),(2,1),(3,2)$\\
    $h(x^{(1)})=2+0=2,\; h(x^{(2)})=2+2=4,\; h(x^{(3)})=2+3=5$\\
    $L_{MSE} = \frac{1}{6}[2^2+3^2+3^2]=\frac{22}{6}=\frac{11}{3}$

    \subsubsection{SGD Update (6B)}
    Pick random point $i$. Update:
    \[w \leftarrow w - \alpha \cdot (h_w(x^{(i)}) - y^{(i)}) \begin{bmatrix}1\\x^{(i)}\end{bmatrix}\]
    \textbf{Example:} Random point $i=2$, $\alpha=1/3$.\\
    Residual $= h(x^{(2)})-y^{(2)} = 4-1=3$\\
    $w' = \begin{bmatrix}2\\1\end{bmatrix} - \frac{1}{3}(3)\begin{bmatrix}1\\2\end{bmatrix} = \begin{bmatrix}2\\1\end{bmatrix}-\begin{bmatrix}1\\2\end{bmatrix}=\begin{bmatrix}1\\-1\end{bmatrix}$

    \subsubsection{Linear Dependence \& Convexity (6C)}
    If features are linearly dependent (e.g.\ $x_2=3x_1$): loss is convex (not strictly), so GD may converge to \textit{different weights} but \textit{same loss}. Independent features $\Rightarrow$ strictly convex $\Rightarrow$ unique weights. \textbf{Key}: same loss, possibly different weights.
    $A: x \rightarrow h^A, x' \rightarrow h^{A'}$ $B: x \rightarrow h^B, x' \rightarrow h^{B'}$
    training on x (linearly dependent) may produce different weights but same loss. training on x' (linearly independent) will produce same weights same loss

    \subsubsection{Feature Scaling for Stability (6D)}
    When features have very different scales (e.g.\ $x_3\sim1000$ vs others $\sim1$):
    \begin{itemize}
        \item \textbf{Standardization}: $(x - \mu)/\sigma$ \quad \checkmark
        \item \textbf{Min-max scaling}: $(x-x_{min})/(x_{max}-x_{min})$ \quad \checkmark
        \item Normal equation fails if features are linearly dependent (non-invertible $X^\top X$)
    \end{itemize}

    \subsubsection{Logistic Regression (6E)}
    Use $\sigma(z)=1/(1+e^{-z})$ when predicting \textit{probability}.\\
    Need bias $w_0$ if $P>0.5$ when all $x_i=0$ (since $\sigma(0)=0.5$, need $w_0>0$).\\
    Write as $P(Y|X) = \sigma(w_0+w_1x_1+w_2x_2+w_3x_3)$. Note: $P(Y|X)$ not $P(Y|w)$.

    \subsubsection{Normal Equation (6F)}
    $w = (X^\top X)^{-1} X^\top y$\\
    Diagonal matrix inverse: flip each diagonal entry $d_i \to 1/d_i$.\\
    \textbf{Example:} $X=\begin{bmatrix}1&0\\0&2\end{bmatrix}$, $y=\begin{bmatrix}5\\6\end{bmatrix}$\\
    $X^\top X = \begin{bmatrix}1&0\\0&4\end{bmatrix}$, $(X^\top X)^{-1}=\begin{bmatrix}1&0\\0&1/4\end{bmatrix}$\\
    $X^\top y = \begin{bmatrix}5\\12\end{bmatrix}$, $w = \begin{bmatrix}1&0\\0&1/4\end{bmatrix}\begin{bmatrix}5\\12\end{bmatrix}=\begin{bmatrix}5\\3\end{bmatrix}$

    \subsubsection{Inverting 2x2 Matrix}
    $
        \text{For } A = \begin{bmatrix}a&b\\c&d\end{bmatrix}, \quad
        A^{-1} = \frac{1}{ad-bc}\begin{bmatrix}d&-b\\-c&a\end{bmatrix}
    $

    \subsubsection{distance of point $x$ from some hyperplane $P$ (or $B$ for boundary)}
    \begin{itemize}
        \item $
                  d = \frac{w^Tx + b}{||w||}
              $, OR
        \item \begin{enumerate}
                  \item find point $z$ on $P$ ($
                            \begin{matrix}
                                x \\ y
                            \end{matrix}
                            \cdot
                            \begin{matrix}
                                w_1 w_2
                            \end{matrix}
                            + b = 0
                        $)
                  \item get $
                            \vec{u} = \vec{x} - \vec{z}
                        $
                  \item find $
                            \text{Proj}_w\vec{u} = \frac{\vec{u} \cdot \vec{w}}{||\vec{w}||}
                        $
              \end{enumerate}
    \end{itemize}
